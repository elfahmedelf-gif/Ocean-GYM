   103→function todayDateInput(): string {
   104→  const d = new Date();
   105→  const offset = d.getTimezoneOffset();
   106→  const local = new Date(d.getTime() - offset * 60 * 1000);
   107→  return local.toISOString().slice(0, 10);
   108→}
   109→
   110→const EMPTY_PROGRESS: ProgressFormState = {
   111→  date: "",
   112→  weight: "",
   113→  fatPercent: "",
   114→  muscleMass: "",
   115→  waist: "",
   116→  chest: "",
   117→  arm: "",
   118→  notes: "",
   119→};
   120→
   121→function toDateInputValue(iso: string | null): string {
   122→  if (!iso) return "";
   123→  const d = new Date(iso);
   124→  if (Number.isNaN(d.getTime())) return "";
   125→  return d.toISOString().slice(0, 10);
   126→}
   127→
   128→function formatDate(iso: string | null): string {
   129→  if (!iso) return "—";
   130→  try {
   131→    return new Date(iso).toLocaleDateString("ar-EG", {
   132→      year: "numeric",
   133→      month: "long",
   134→      day: "numeric",
   135→    });
   136→  } catch {
   137→    return iso;
   138→  }
   139→}
   140→
   141→function formatLogDate(iso: string): string {
   142→  try {
   143→    return new Date(iso).toLocaleDateString("ar-EG", {
   144→      year: "numeric",
   145→      month: "short",
   146→      day: "numeric",
   147→    });
   148→  } catch {
   149→    return iso;
   150→  }
   151→}
   152→
   153→function fmtNum(v: number | null): string {
   154→  if (v === null || v === undefined) return "—";
   155→  return String(v);
   156→}

   272→  async function handleSave(e: React.FormEvent<HTMLFormElement>) {
   273→    e.preventDefault();
   274→
   275→    if (!form.name.trim() || !form.email.trim() || !form.phone.trim()) {
   276→      toast.error("الاسم والبريد الإلكتروني ورقم الهاتف مطلوبة");
   277→      return;
   278→    }
   279→
   280→    setSaving(true);
   281→    const payload = {
   282→      name: form.name.trim(),
   283→      email: form.email.trim(),
   284→      phone: form.phone.trim(),
   285→      planId: form.planId === "none" ? null : form.planId,
   286→      status: form.status,
   287→      startDate: form.startDate || null,
   288→      endDate: form.endDate || null,
   289→      notes: form.notes.trim() || null,
   290→      nutritionPlanId: form.nutritionPlanId === "none" ? null : form.nutritionPlanId,
   291→      accessCode: form.accessCode.trim() || null,
   292→    };
   293→
   294→    try {
   295→      if (editing) {
   296→        await jsonPut(`/api/admin/members/${editing.id}`, payload);
   297→        toast.success("تم تحديث بيانات العضو بنجاح");
   298→      } else {
   299→        await jsonPost("/api/admin/members", payload);
   300→        toast.success("تم إضافة العضو بنجاح");
   301→      }
   302→      setFormOpen(false);
   303→      await loadMembers();
   304→    } catch (err) {
   305→      toast.error(err instanceof Error ? err.message : "تعذر الحفظ");
   306→    } finally {
   307→      setSaving(false);
   308→    }
   309→  }

   363→  async function handleSaveProgress(e: React.FormEvent<HTMLFormElement>) {
   364→    e.preventDefault();
   365→    if (!progressMember) return;
   366→
   367→    if (!progressForm.date) {
   368→      toast.error("تاريخ القياس مطلوب");

   372→    if (!progressForm.weight || Number.isNaN(weight) || weight <= 0) {
   373→      toast.error("الوزن مطلوب ويجب أن يكون أكبر من صفر");
   374→      return;
   375→    }
   376→
   377→    setProgressSaving(true);
   378→    const payload = {
   379→      date: new Date(`${progressForm.date}T12:00:00`).toISOString(),
   380→      weight,
   381→      fatPercent: progressForm.fatPercent ? Number(progressForm.fatPercent) : null,
   382→      muscleMass: progressForm.muscleMass ? Number(progressForm.muscleMass) : null,
   383→      waist: progressForm.waist ? Number(progressForm.waist) : null,
   384→      chest: progressForm.chest ? Number(progressForm.chest) : null,
   385→      arm: progressForm.arm ? Number(progressForm.arm) : null,
   386→      notes: progressForm.notes.trim() || null,
   387→    };
   388→
   389→    try {
   390→      await jsonPost(`/api/admin/members/${progressMember.id}/progress`, payload);
   391→      toast.success("تم إضافة سجل المتابعة بنجاح");
   392→      setProgressForm({ ...EMPTY_PROGRESS, date: todayDateInput() });
   393→      const logs = await apiFetch<ProgressLogItem[]>(
   394→        `/api/admin/members/${progressMember.id}/progress`
   395→      );
   396→      setProgressLogs(logs);
   397→      await loadMembers();
   398→    } catch (err) {
   399→      toast.error(err instanceof Error ? err.message : "تعذر حفظ السجل");
   400→    } finally {
   401→      setProgressSaving(false);
   402→    }
   403→  }
